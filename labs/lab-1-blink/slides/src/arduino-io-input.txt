void setup() {
  // set pin 3 to an input
  pinMode(3, INPUT);
  // when the input value of pin 3 falls
  // call the function 'wheelInterrupt'
  attachInterrupt(
    digitalPinToInterrupt(3), wheelInterrupt, FALLING
  );
}
