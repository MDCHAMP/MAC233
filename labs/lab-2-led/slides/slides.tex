\begin{frame}[fragile]{Reminder: `pinMode' and `digitalWrite'}
    \normalsize
    
    \verb|pinMode| sets the IO direction of a pin and belongs in \verb|setup|. 
    
    \vspace{.5em}

    \verb|digitalWrite| then sets the pin HIGH (+5V) or LOW (0V).
    
    \vspace{.5em}
    
\begin{minted}[fontsize=\small]{arduino}
#define LED_PIN 2

void setup() {
  pinMode(LED_PIN, OUTPUT);     // once, in setup
}

void loop() {
  digitalWrite(LED_PIN, HIGH);  // 5 V, LED on
  delay(1000);
  digitalWrite(LED_PIN, LOW);   // 0 V, LED off
  delay(1000);
}
\end{minted}
\end{frame}

\begin{frame}[fragile]{What `HIGH' actually does}
    \begin{center}
        \begin{tikzpicture}[
            wire/.style={line width=.6mm, draw=black},
            lbl/.style={font=\small, align=center},
        ]
            % the pin
            \node[lbl, anchor=east] at (-0.15, 0) {Pin D2\\\textbf{HIGH} $= 5$\,V};
            \draw[wire] (0, 0) -- (1.3, 0);

            % resistor
            \draw[wire, fill=white] (1.3, -0.22) rectangle (2.6, 0.22);
            \node[lbl] at (1.95, 0.6) {$220\,\Omega$};
            \draw[wire] (2.6, 0) -- (3.9, 0);

            % LED
            \filldraw[draw=black, fill=red!70, line width=.5mm]
                (3.9, -0.38) -- (3.9, 0.38) -- (4.6, 0) -- cycle;
            \draw[wire] (4.6, -0.38) -- (4.6, 0.38);
            \node[lbl] at (4.25, 0.75) {LED};
            \draw[wire] (4.6, 0) -- (6.0, 0);

            % ground symbol
            \draw[wire] (6.0, -0.34) -- (6.0, 0.34);
            \draw[wire] (6.2, -0.20) -- (6.2, 0.20);
            \draw[wire] (6.4, -0.08) -- (6.4, 0.08);
            \node[lbl, anchor=west] at (6.6, 0) {GND\\$0$\,V};

            % current
            \draw[-latex, purple, line width=.4mm] (0.7, -0.95) -- (5.5, -0.95);
            \node[lbl, color=purple] at (3.1, -1.35) {current flows from 5\,V down to 0\,V};
        \end{tikzpicture}
    \end{center}

    \vspace{.5em}
    \verb|digitalWrite(LED_PIN, HIGH)| connects the pin to 5\,V, and `LOW' connects it to 0\,V. It is that 5\,V that pushes a current through the LED and lights it.

    \vspace{.75em}
    The LED itself uses up about 2\,V, so the resistor takes the other 3\,V. With no resistor, nothing limits the current and both the LED and the pin can be damaged.

\end{frame}
